"""Minimal licensed-real-timbre qizi accompaniment for backstage_v1."""

from __future__ import annotations

import json
from functools import lru_cache
from pathlib import Path

from .pipeline import AccompanimentError, PENTATONIC, infer_tonic
from .traditional import PARTS, Arrangement, PerformanceEvent, export_traditional_midi


ARRANGEMENT_VERSION = "backstage_v1.1.1_legato"
TIMBRE_STATUS = "licensed_free_substitute"
EXPERT_REVIEW_STATUS = "not_requested"


@lru_cache(maxsize=1)
def load_qizi_template():
    path = Path(__file__).with_name("templates") / "backstage_v1" / "qizi.json"
    if not path.is_file():
        raise AccompanimentError("backstage_v1 qizi template is missing")
    return json.loads(path.read_text(encoding="utf-8"))


def _nearest_scale(target, tonic, low, high):
    allowed = [pitch for pitch in range(low, high + 1) if (pitch - tonic) % 12 in PENTATONIC]
    return min(allowed, key=lambda pitch: (abs(pitch - target), pitch))


def _fit_range(pitch, low, high):
    while pitch < low:
        pitch += 12
    while pitch > high:
        pitch -= 12
    return max(low, min(high, pitch))


def _event(start, end, pitch, velocity, technique, pitch_curve=()):
    return PerformanceEvent(float(start), max(float(start) + 0.001, float(end)), int(pitch),
                            int(velocity), technique, tuple(pitch_curve))


def _lead(timeline, tonic, style):
    velocity = {"gentle": 43, "cheerful": 49, "intense": 55}[style]
    result = []
    phrase_ids = []
    for source in timeline:
        if source.midi is not None and source.phrase_id not in phrase_ids:
            phrase_ids.append(source.phrase_id)
    for phrase_id in phrase_ids:
        phrase = [source for source in timeline
                  if source.midi is not None and source.phrase_id == phrase_id]
        if not phrase:
            continue
        anchors = [phrase[0]]
        for index, source in enumerate(phrase[1:-1], start=1):
            previous = phrase[index - 1]
            following = phrase[index + 1]
            direction_before = source.midi - previous.midi
            direction_after = following.midi - source.midi
            direction_change = direction_before * direction_after < 0
            spaced = source.start - anchors[-1].start >= 0.72
            expressive = source.duration >= 0.68 or source.phrase_end
            if spaced or expressive or (direction_change and source.start - anchors[-1].start >= 0.32):
                anchors.append(source)
        if phrase[-1] is not anchors[-1]:
            anchors.append(phrase[-1])
        for index, source in enumerate(anchors):
            end = anchors[index + 1].start if index + 1 < len(anchors) else phrase[-1].end
            pitch = _nearest_scale(_fit_range(source.midi, 59, 81), tonic, 59, 81)
            curve = ((0.0, 0.0), (0.38, 0.10), (0.62, -0.10), (1.0, 0.0))
            if index == len(anchors) - 1:
                curve = ((0.0, 0.0), (0.78, -0.35), (1.0, 0.0))
            result.append(_event(source.start, end, pitch, velocity, "sustain", curve))
    return result


def _lead_coverage_ratio(timeline, events):
    phrases = {}
    for source in timeline:
        if source.midi is not None:
            start, end = phrases.get(source.phrase_id, (source.start, source.end))
            phrases[source.phrase_id] = (min(start, source.start), max(end, source.end))
    singing = sum(end - start for start, end in phrases.values())
    covered = 0.0
    for start, end in phrases.values():
        intervals = sorted((max(start, event.start), min(end, event.end)) for event in events
                           if max(start, event.start) < min(end, event.end))
        cursor = start
        for left, right in intervals:
            if right <= cursor:
                continue
            covered += right - max(left, cursor)
            cursor = max(cursor, right)
    return covered / singing if singing else 0.0


def _sparse_plucks_under_voice(timeline, tonic, style):
    velocity = {"gentle": 31, "cheerful": 36, "intense": 41}[style]
    result = []
    previous_phrase = None
    for source in timeline:
        if source.midi is None:
            continue
        phrase_start = source.phrase_id != previous_phrase
        previous_phrase = source.phrase_id
        if not phrase_start and source.duration < 0.82:
            continue
        pitch = _nearest_scale(source.midi - 12, tonic, 56, 72)
        technique = "tremolo" if source.duration >= 0.82 else "short_pluck"
        duration = min(source.duration * 0.72, 0.72 if technique == "tremolo" else 0.13)
        result.append(_event(source.start, source.start + duration, pitch, velocity, technique))
    return result


def _template_pitch(source_pitch, degree, tonic, part):
    if part == "wind":
        return _nearest_scale(source_pitch + int(degree), tonic, 69, 87)
    return _nearest_scale(source_pitch + int(degree), tonic, 56, 72)


def _intro(timeline, tonic, style, template):
    parts = {part: [] for part in PARTS}
    first = next((source for source in timeline if source.midi is not None), None)
    if first is None or first.start < 0.18:
        return parts, [], ["intro skipped because the initial rest is shorter than 0.18 s"]
    lyric_signature = sum(ord(char) for source in timeline for char in source.lyric if source.midi is not None)
    variant = template["intros"][lyric_signature % len(template["intros"])]
    scale = {"gentle": 0.9, "cheerful": 1.0, "intense": 1.1}[style]
    for spec in variant["events"]:
        start = float(spec["offset"])
        end = min(first.start, start + float(spec["duration"]))
        if end <= start:
            continue
        pitch = spec.get("pitch")
        if pitch is None:
            pitch = _template_pitch(first.midi, spec.get("degree", 0), tonic, spec["part"])
        parts[spec["part"]].append(_event(start, end, pitch, round(spec["velocity"] * scale), spec["technique"]))
    return parts, [variant["id"]], []


def _interludes(timeline, tonic, style, template):
    parts = {part: [] for part in PARTS}
    selected_ids, warnings = [], []
    scale = {"gentle": 0.9, "cheerful": 1.0, "intense": 1.1}[style]
    variants = template["interludes"]
    last_pitched = next((source for source in reversed(timeline) if source.midi is not None), None)
    for source in timeline:
        if source.midi is None or not source.phrase_end or not source.interlude_window:
            continue
        if source is last_pitched:
            continue
        window_start, window_end = source.interlude_window
        gap = window_end - window_start
        preferred = variants[(source.phrase_id or 0) % len(variants)]
        usable = [item for item in variants if gap >= float(item["minimum_gap"])]
        if not usable:
            continue
        variant = preferred if preferred in usable else usable[-1]
        selected_ids.append(variant["id"])
        for spec in variant["events"]:
            if "offset_from_end" in spec:
                start = window_end - float(spec["offset_from_end"])
            else:
                start = window_start + float(spec.get("offset", 0.0))
            start = max(window_start, start)
            end = min(window_end, start + float(spec["duration"]))
            if end <= start:
                continue
            pitch = spec.get("pitch")
            if pitch is None:
                pitch = _template_pitch(source.midi, spec.get("degree", 0), tonic, spec["part"])
            parts[spec["part"]].append(_event(start, end, pitch, round(spec["velocity"] * scale), spec["technique"]))
    return parts, selected_ids, warnings


def _ending(timeline, tonic, style, template):
    parts = {part: [] for part in PARTS}
    phrase_count = len({source.phrase_id for source in timeline if source.phrase_id is not None})
    variant = template["endings"][max(0, phrase_count - 1) % len(template["endings"])]
    scale = {"gentle": 0.9, "cheerful": 1.0, "intense": 1.1}[style]
    last_pitched = next((source for source in reversed(timeline) if source.midi is not None), None)
    if last_pitched is None:
        return parts, []
    total = timeline[-1].end
    outro_start = timeline[-1].start if timeline[-1].midi is None else last_pitched.end
    available = total - outro_start
    if available <= 0.18:
        return parts, []
    melodic_end = max(outro_start + 0.12, total - 0.45)
    midpoint = min(melodic_end, outro_start + min(0.92, available * 0.42))
    first_pitch = _nearest_scale(last_pitched.midi + 2, tonic, 59, 81)
    final_pitch = _nearest_scale(last_pitched.midi, tonic, 59, 81)
    if midpoint > outro_start + 0.03:
        parts["lead_bowed"].append(_event(
            outro_start + 0.02, midpoint, first_pitch, round(42 * scale), "sustain",
            ((0.0, 0.0), (0.55, 0.12), (1.0, 0.0)),
        ))
    if melodic_end > midpoint:
        parts["lead_bowed"].append(_event(
            midpoint, melodic_end, final_pitch, round(40 * scale), "sustain",
            ((0.0, 0.0), (0.72, -0.28), (1.0, -0.45)),
        ))
    plucked_pitch = _nearest_scale(last_pitched.midi - 12, tonic, 56, 72)
    plucked_end = max(outro_start + 0.16, total - 0.52)
    parts["plucked"].append(_event(
        outro_start + 0.05, plucked_end, plucked_pitch,
        round(29 * scale), "tremolo",
    ))
    latest_template_end = max(float(spec["offset"]) + float(spec["duration"])
                              for spec in variant["events"])
    template_shift = max(outro_start, total - 0.05 - latest_template_end)
    for spec in variant["events"]:
        start = template_shift + float(spec["offset"])
        end = min(total, start + float(spec["duration"]))
        if end > start:
            parts[spec["part"]].append(_event(
                start, end, spec["pitch"], round(spec["velocity"] * scale), spec["technique"]))
    return parts, [variant["id"]]


def _merge(target, source):
    for part in PARTS:
        target[part].extend(source[part])


def arrange_backstage_v1(timeline, qupai="qizi", style="gentle"):
    if qupai != "qizi":
        raise AccompanimentError("backstage_v1 currently supports qizi only")
    if style not in {"gentle", "cheerful", "intense"}:
        raise AccompanimentError(f"unknown accompaniment style: {style}")
    if not timeline:
        raise AccompanimentError("backstage_v1 requires a non-empty timeline")
    template = load_qizi_template()
    tonic = infer_tonic(timeline)
    parts = {part: [] for part in PARTS}
    parts["lead_bowed"] = _lead(timeline, tonic, style)
    parts["plucked"] = _sparse_plucks_under_voice(timeline, tonic, style)
    intro, intro_ids, warnings = _intro(timeline, tonic, style, template)
    interludes, interlude_ids, interlude_warnings = _interludes(timeline, tonic, style, template)
    ending, ending_ids = _ending(timeline, tonic, style, template)
    _merge(parts, intro)
    _merge(parts, interludes)
    _merge(parts, ending)
    for part in PARTS:
        parts[part].sort(key=lambda item: (item.start, item.end, item.pitch, item.technique))
    arrangement = Arrangement(
        parts=parts,
        template_version=template["template_version"],
        review_status="unreviewed",
        qupai="qizi",
        fallback_reason="",
        warnings=[*warnings, *interlude_warnings],
    )
    metadata = {
        "arrangement_version": ARRANGEMENT_VERSION,
        "timbre_status": TIMBRE_STATUS,
        "expert_review_status": EXPERT_REVIEW_STATUS,
        "template_ids": [*intro_ids, *interlude_ids, *ending_ids],
        "lead_strategy": "phrase_anchor_legato_v1",
        "lead_coverage_ratio": round(_lead_coverage_ratio(timeline, parts["lead_bowed"]), 6),
        "outro_window": {
            "start": timeline[-1].start if timeline[-1].midi is None else timeline[-1].end,
            "end": timeline[-1].end,
        },
    }
    return arrangement, metadata


def export_backstage_v1_midi(timeline, arrangement, output_path, render_path=None):
    return export_traditional_midi(timeline, arrangement, output_path, render_path)
