"""Orchestration entrypoint for the research_hybrid_v1 profile."""

from __future__ import annotations

from dataclasses import dataclass
from typing import Any, Dict, Iterable, Mapping, Optional, Sequence, Tuple

from .candidates import ArrangementCandidate, CandidateProvider, RuleCandidateProvider
from .conditions import ConditionBundle, condition_bundle_from_request, condition_bundle_from_timeline
from .constraints import CandidateEvaluation, evaluate_candidates, rank_candidates
from .evaluation import TechnicalEvaluation, evaluate_technical_alignment
from .pipeline import AccompanimentError
from .structure import OperaTimeline, build_opera_timeline


RESEARCH_PROFILE = "research_hybrid_v1"
RESEARCH_SCHEMA_VERSION = 1
RESEARCH_IMPLEMENTATION_VERSION = "research_hybrid_v1.0"


@dataclass(frozen=True)
class ResearchGenerationResult:
    selected: ArrangementCandidate
    evaluations: Tuple[CandidateEvaluation, ...]
    timeline: OperaTimeline
    technical_evaluation: TechnicalEvaluation
    seed: int
    provider_name: str

    def metadata(self, include_timeline: bool = True, include_candidate_events: bool = False) -> Dict[str, Any]:
        ranked = list(self.evaluations)
        selected_rank = next(index for index, item in enumerate(ranked)
                             if item.candidate_id == self.selected.candidate_id) + 1
        return {
            "schema_version": RESEARCH_SCHEMA_VERSION,
            "profile": RESEARCH_PROFILE,
            "implementation_version": RESEARCH_IMPLEMENTATION_VERSION,
            "review_status": "unreviewed",
            "expert_review_status": "not_requested",
            "claims_status": "technical_framework_only",
            "seed": self.seed,
            "provider": self.provider_name,
            "condition_summary": self.timeline.conditions.summary(),
            "structure": self.timeline.as_dict(include_units=False) if include_timeline else None,
            "candidate_count": len(ranked),
            "candidate_ranking": [evaluation.as_dict() for evaluation in ranked],
            "selected_candidate": self.selected.as_dict(include_events=include_candidate_events),
            "selected_rank": selected_rank,
            "technical_evaluation": self.technical_evaluation.as_dict(),
            "warnings": [
                *[dict(warning) for warning in self.timeline.warnings],
                {"code": "not_expert_validated", "level": "warning",
                 "message": "technical constraints do not establish Gezi-opera authenticity"},
                {"code": "not_a_learned_model", "level": "info",
                 "message": "this release provides a rule candidate baseline and model interface only"},
            ],
        }


def _normalize_seed(seed: Any) -> int:
    try:
        value = int(seed)
    except (TypeError, ValueError) as exc:
        raise AccompanimentError("research seed must be an integer") from exc
    if not -(2 ** 63) <= value < 2 ** 63:
        raise AccompanimentError("research seed must fit in signed 64-bit range")
    return value


def _condition_bundle(
    legacy_timeline: Optional[Sequence[Any]],
    conditions: Optional[Any],
    qupai: str,
) -> ConditionBundle:
    if isinstance(conditions, ConditionBundle):
        bundle = conditions
    elif isinstance(conditions, Mapping):
        bundle = condition_bundle_from_request(conditions)
    elif conditions is not None:
        raise AccompanimentError("conditions must be a ConditionBundle or mapping")
    elif legacy_timeline is not None:
        bundle = condition_bundle_from_timeline(legacy_timeline, qupai=qupai)
    else:
        raise AccompanimentError("legacy timeline or conditions are required")
    if bundle.qupai != "qizi":
        raise AccompanimentError("research_hybrid_v1 currently supports qizi only")
    if qupai not in {"qizi", "xiaodiao"} and qupai != bundle.qupai:
        raise AccompanimentError("qupai conflicts with supplied condition bundle")
    return bundle


def build_research_generation(
    legacy_timeline: Optional[Sequence[Any]],
    style: str = "gentle",
    seed: int = 0,
    conditions: Optional[Any] = None,
    provider: Optional[CandidateProvider] = None,
    sample_parts: Optional[Iterable[str]] = None,
    qupai: str = "qizi",
) -> ResearchGenerationResult:
    """Build and rank arrangements without rendering audio."""

    normalized_seed = _normalize_seed(seed)
    bundle = _condition_bundle(legacy_timeline, conditions, qupai)
    timeline = build_opera_timeline(bundle)
    active_provider = provider or RuleCandidateProvider()
    candidates = tuple(active_provider.generate(timeline, style, normalized_seed))
    if not candidates:
        raise AccompanimentError("candidate provider returned no arrangements")
    evaluations = evaluate_candidates(candidates, timeline, style, sample_parts)
    selected, ranked = rank_candidates(candidates, evaluations)
    technical = evaluate_technical_alignment(selected.arrangement, timeline)
    return ResearchGenerationResult(selected, ranked, timeline, technical,
                                    normalized_seed, active_provider.provider_name)


def arrange_research_hybrid(
    legacy_timeline: Optional[Sequence[Any]],
    qupai: str = "qizi",
    style: str = "gentle",
    seed: int = 0,
    conditions: Optional[Any] = None,
    sample_parts: Optional[Iterable[str]] = None,
):
    """Return the selected Arrangement and versioned, explainable metadata."""

    result = build_research_generation(legacy_timeline, style, seed, conditions,
                                       sample_parts=sample_parts, qupai=qupai)
    metadata = result.metadata()
    result.selected.arrangement.warnings = [warning["message"] for warning in metadata["warnings"]]
    return result.selected.arrangement, metadata
